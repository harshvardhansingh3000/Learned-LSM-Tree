% PROBLEM STATEMENT
% =================================================
\begin{frame}{Problem Statement}

\begin{block}{Challenges in Traditional LSM-Trees}
\begin{itemize}
    \item Multiple Bloom filter probes increase lookup latency
    \item Bloom filters consume large memory at deeper levels
    \item Read amplification increases with SSTable growth
    \item Existing systems are difficult to study and modify
\end{itemize}
\end{block}

\vspace{4mm}

\begin{block}{Goal}
Design a compact LSM-tree based key--value engine with:
\begin{itemize}
    \item Efficient write path
    \item Optimised read path
    \item ML-assisted lookup optimisation
    \item Reduced Bloom filter memory overhead
\end{itemize}
\end{block}

\end{frame}
